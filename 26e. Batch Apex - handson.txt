public class BatchApex implements Database.Batchable<SObject> {
    
    public Database.QueryLocator start(Database.BatchableContext BC) {
        system.debug('Started Batch Apex program');
        return Database.getQueryLocator('select firstname, lastname, company, status from Lead');
    }
    
    public void execute(Database.BatchableContext BC, List<Lead> leads) {
        system.debug('In execute phase');
        for(Lead lead: leads) {
            lead.company = 'Amazon';
        }
      update leads;        
    }
    
    
    public void finish(Database.BatchableContext BC) {
        system.debug('Finished Batch Apex program');
    }

}
