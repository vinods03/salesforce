public class FutureExample2 {
    
    @future
    public static void updateRecords(Set<Id> recordId) {
        
        List<Lead> leads = [select firstname, lastname, company, status from lead where id in :recordId];
        for(Lead lead: Leads) {
            lead.company = 'Google';
        }
        
        update leads;
        
    }

}
