File - New - Apex Class:

public class ScheduleApex implements Schedulable {
    
    public void execute(SchedulableContext context) {
        
        List<Lead> leads = [select firstname, lastname, company, status from lead];
        
        for(Lead lead: Leads) {
            lead.company = 'Amazon';
        }
        
        update leads;
    }

}

Debug: Open execute anonymous window

ScheduleApex sa = new ScheduleApex();
String cronExp = '0 30 2 ? * MON-FRI';
String jobId = System.schedule('Sch Job Example', cronExp, sa);
System.debug('Job ID:'+ jobId);  // Note down the jobId
