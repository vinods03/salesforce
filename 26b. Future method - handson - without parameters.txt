public class FutureExample1 {
    
    @future
    public static void updateRecords() {
        
        List<Lead> records = [select firstname, lastname, company, status from lead where firstname = 'Vinod'];
        for (Lead record: records) {
            record.status = 'Working - Contacted';
        }
        
        update records;
        
    }
    
    @future
    public static void insertRecords() {
        Lead record = new Lead(FirstName='Axar', LastName='Patel',Company='CWC', status = 'Open - Not Contacted');
        insert record;
    }

}
